# --- Orchestration (Phase 2) ---
# Pulls langchain-core (>=1.4.7,<2) transitively. That is the only LangChain
# package in the project and nothing imports from it directly.
langgraph==1.2.11

# --- LLM provider (Phase 2) ---
# Single provider by design. Chosen for native structured output
# (response_schema + response.parsed) and real usage_metadata token counts.
# A second provider would be added as another adapter behind LLMProvider,
# never as a direct import inside an agent.
google-genai==2.20.0
# Second provider behind the same LLMProvider protocol. Its free tier allows
# far more requests per day, which is what makes repeated runs practical.
# Every dependency it needs is already in the tree.
groq==1.7.0

# --- Research provider (Phase 3) ---
# Tavily's REST API is a single authenticated POST, so it is called directly
# rather than through tavily-python, which would pull in requests and tiktoken
# for no benefit. httpx is already in the tree via google-genai; pinned here
# because we now depend on it directly.
httpx==0.28.1
